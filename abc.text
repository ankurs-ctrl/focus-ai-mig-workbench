5. Check Generated Inferences
SELECT
    i.INFERENCE_ID,
    i.INFERENCE_GOAL_ID,
    i.INFERENCE_TYPE,
    i.SUBJECT_KEY,
    i.SUMMARY,
    i.CONFIDENCE,
    i.CONFIDENCE_BAND,

    i.CREATED_AT
FROM TBL_WORKBENCH_INFERENCES i
JOIN TBL_AGENT_FIR_360 f
    ON f.INFERENCE_ID = i.INFERENCE_ID
WHERE f.FEEDBACK_PAYLOAD:sql_asset_id::STRING =
        '3abf76268adb34d1cb6caf9388e572de'
   OR f.CONTEXT_KEY =
        'ctx_003c4873ac994cc841c88307f02ce73b869f1a75'
   OR f.STTM_ID = '401'
ORDER BY i.CREATED_AT DESC;
