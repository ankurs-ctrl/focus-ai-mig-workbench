def _quote_identifier(identifier: str) -> str:
    return '"' + identifier.replace('"', '""') + '"'


def _activate_session_role(session: Session, role: Optional[str]) -> None:
    normalized_role = (role or "").strip()
    if not normalized_role:
        return
    session.sql(f"USE ROLE {_quote_identifier(normalized_role)}").collect()
    logger.info("Activated Snowflake session role: %s", normalized_role)
