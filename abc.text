### 2.7 — Three new helper methods (180 lines)

Baseline line ~733, after `_connected_join_plan` returns (`return connected, ordered`) and **before** `_ordered_cte_nodes`:

```python
    @staticmethod
    def _expression_relation_ids(expression: str, alias_to_id: dict[str, str]) -> set[str]:
        """Relations an expression reads from, resolved through their SQL aliases."""
        scrubbed = re.sub(r"'(?:''|[^'])*'", "''", expression)
        found: set[str] = set()
        for qualifier, _column in re.findall(
            r"(?<![$\w])([A-Za-z_][A-Za-z0-9_$]*)\s*\.\s*([A-Za-z_][A-Za-z0-9_$]*)",
            scrubbed,
        ):
            relation_id = alias_to_id.get(qualifier.upper())
            if relation_id is not None:
                found.add(relation_id)
        return found

    @staticmethod
    def _derived_source_aliases(node: RelationNode) -> dict[str, str]:
        """SQL alias -> qualified table name, read from the derived source's own SQL.

        Column lineage is recorded against the SQL aliases the derived source used
        ("c.ID"), so those aliases have to be resolved back to real tables before
        lineage can be attributed to a parent relation.
        """
        if not node.sql_text:
            return {}
        try:
            statement = parse_one(node.sql_text, read="snowflake")
        except Exception:
            return {}
        alias_map: dict[str, str] = {}
        for table in statement.find_all(exp.Table):
            qualified = ".".join(
                part for part in (table.catalog, table.db, table.name) if part
            ).upper()
            alias = str(table.alias_or_name or "").strip().upper()
            if alias and qualified:
                alias_map[alias] = qualified
        return alias_map

    @classmethod
    def _derived_output_column_for_parent(
        cls,
        node: RelationNode,
        base_id: str,
        parent_column: str,
    ) -> str | None:
        """Name the derived source exposes for a parent table's join column.

        Derived sources routinely rename columns ("c.ID AS contact_id"), so the
        parent's column name is not usable directly. Returns None whenever the
        column cannot be attributed unambiguously, which makes the caller leave
        the relation disconnected rather than join on the wrong column.
        """
        target = str(parent_column or "").strip().upper()
        if not target:
            return None
        available = cls._relation_output_columns(node)
        base_upper = base_id.upper()
        base_short = base_upper.rsplit(".", 1)[-1]
        lineage = [
            item
            for item in node.column_semantics
            if isinstance(item, dict) and item.get("source_columns")
        ]
        if lineage:
            alias_map = cls._derived_source_aliases(node)

            def qualifier_is_base(qualifier: str) -> bool:
                resolved = alias_map.get(qualifier)
                if resolved is not None:
                    return resolved == base_upper or resolved.rsplit(".", 1)[-1] == base_short
                # Unaliased references appear as the table's own name.
                return qualifier in {base_upper, base_short}

            matches: set[str] = set()
            for item in lineage:
                name = str(item.get("name") or "").strip().upper()
                if not name or name not in available:
                    continue
                for source in item.get("source_columns") or []:
                    text = str(source or "").strip().upper()
                    qualifier, separator, column = text.rpartition(".")
                    if separator:
                        if column == target and qualifier_is_base(qualifier):
                            matches.add(name)
                    elif text == target and len(node.base_relation_ids) == 1:
                        matches.add(name)
            # Two outputs carrying the same source column is ambiguous, not a match.
            return next(iter(matches)) if len(matches) == 1 else None
        # No column-level lineage recorded. A bare name match is only unambiguous
        # when the derived source has exactly one parent, otherwise an identically
        # named column from a different parent could be picked.
        if len(node.base_relation_ids) == 1 and target in available:
            return target
        return None

    @classmethod
    def _lineage_inherited_edges(
        cls,
        *,
        disconnected_ids: set[str],
        nodes: dict[str, RelationNode],
        edges: list[Any],
        connected_ids: set[str],
        existing_edge_ids: set[str],
    ) -> list[RelationEdge]:
        """Re-project a parent table's join onto a derived source built from it.

        Only ever called for relations that are already disconnected, and only
        emits an edge when the derived source still exposes every join column the
        parent used. A derived source that projected those columns away yields
        nothing, so the caller fails with its original error rather than joining
        on a guess - silently wrong rows are worse than a refusal to compile.
        """
        inherited: list[RelationEdge] = []
        for relation_id in sorted(disconnected_ids):
            node = nodes.get(relation_id)
            if node is None or node.kind != RelationNodeKind.DERIVED_SOURCE:
                continue
            if not cls._relation_output_columns(node):
                continue
            for base_id in node.base_relation_ids:
                if base_id == relation_id or base_id not in nodes:
                    continue
                for edge in edges:
                    if edge.left_relation_id == base_id:
                        other_id, base_on_left = edge.right_relation_id, True
                    elif edge.right_relation_id == base_id:
                        other_id, base_on_left = edge.left_relation_id, False
                    else:
                        continue
                    # The far side must already reach the driving relation,
                    # otherwise the inherited edge would not connect anything.
                    if other_id == relation_id or other_id not in connected_ids:
                        continue
                    if not edge.conditions:
                        continue
                    # Rewrite the parent's side of every condition to the name the
                    # derived source actually exposes. If any single column cannot
                    # be resolved the whole edge is abandoned - a partially
                    # translated join would silently match the wrong rows.
                    rewritten: list[RelationshipConditionItem] = []
                    for condition in edge.conditions:
                        parent_column = (
                            condition.left_column if base_on_left else condition.right_column
                        )
                        derived_column = cls._derived_output_column_for_parent(
                            node, base_id, parent_column
                        )
                        if derived_column is None:
                            rewritten = []
                            break
                        rewritten.append(
                            RelationshipConditionItem(
                                left_column=derived_column if base_on_left else condition.left_column,
                                operator=condition.operator,
                                right_column=condition.right_column if base_on_left else derived_column,
                            )
                        )
                    if not rewritten:
                        continue
                    edge_id = f"lineage:{relation_id}:{edge.edge_id}"
                    if edge_id in existing_edge_ids:
                        continue
                    existing_edge_ids.add(edge_id)
                    inherited.append(
                        RelationEdge(
                            edge_id=edge_id,
                            # The derived source takes the parent's own side of
                            # the join so an outer join keeps its direction.
                            left_relation_id=relation_id if base_on_left else other_id,
                            right_relation_id=other_id if base_on_left else relation_id,
                            join_type=edge.join_type,
                            conditions=rewritten,
                            additional_predicate=edge.additional_predicate,
                            provenance="DERIVED_LINEAGE",
                            validation_status=edge.validation_status,
                        )
                    )
        return inherited

    @staticmethod
    def _ordered_cte_nodes(nodes: list[RelationNode], required_ids: set[str]) -> list[RelationNode]:
```
