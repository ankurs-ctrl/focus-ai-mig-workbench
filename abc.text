Hi Team, can you please help run the below Snowflake admin setup for our SPCS app?
We need this because the app now uses Snowflake OAuth so signed-in users can securely call Cortex Agent/Analyst REST APIs from SPCS. The secrets are stored in Snowflake so no OAuth credentials are hardcoded in code or service specs.
Please replace the placeholders before running.
CREATE OR REPLACE SECURITY INTEGRATION <OAUTH_INTEGRATION_NAME>
  TYPE = OAUTH
  ENABLED = TRUE
  OAUTH_CLIENT = CUSTOM
  OAUTH_CLIENT_TYPE = 'CONFIDENTIAL'
  OAUTH_REDIRECT_URI = 'https://<CLIENT_SPCS_APP_HOST>/api/v1/auth/callback'
  OAUTH_ALTERNATE_REDIRECT_URIS = (
    'http://localhost:3000/api/v1/auth/callback'
  )
  OAUTH_ISSUE_REFRESH_TOKENS = TRUE
  OAUTH_REFRESH_TOKEN_VALIDITY = 86400
  OAUTH_ENFORCE_PKCE = TRUE
  OAUTH_USE_SECONDARY_ROLES = IMPLICIT
  PRE_AUTHORIZED_ROLES_LIST = (
    '<APP_ADMIN_ROLE>',
    '<APP_PUBLISHER_ROLE>',
    '<APP_VIEWER_ROLE>'
  )
  BLOCKED_ROLES_LIST = (
    'ACCOUNTADMIN',
    'SECURITYADMIN',
    'ORGADMIN',
    'GLOBALORGADMIN'
  );

DESC SECURITY INTEGRATION <OAUTH_INTEGRATION_NAME>;

SELECT SYSTEM$SHOW_OAUTH_CLIENT_SECRETS('<OAUTH_INTEGRATION_NAME>');
Then use the returned OAuth client ID/secret here:
USE DATABASE <APP_DB>;
USE SCHEMA <APP_SCHEMA>;

CREATE OR REPLACE SECRET <APP_DB>.<APP_SCHEMA>.STTM_BUILDER_OAUTH_CLIENT_CREDENTIALS
  TYPE = PASSWORD
  USERNAME = '<OAUTH_CLIENT_ID_FROM_SYSTEM_SHOW>'
  PASSWORD = '<OAUTH_CLIENT_SECRET_FROM_SYSTEM_SHOW>';

CREATE OR REPLACE SECRET <APP_DB>.<APP_SCHEMA>.STTM_BUILDER_OAUTH_SESSION_KEYS
  TYPE = PASSWORD
  USERNAME = '<RANDOM_64_CHAR_SESSION_SECRET>'
  PASSWORD = '<RANDOM_32_OR_64_CHAR_ENCRYPTION_KEY>';
And grant the SPCS service owner role access:
GRANT READ ON SECRET <APP_DB>.<APP_SCHEMA>.STTM_BUILDER_OAUTH_CLIENT_CREDENTIALS
  TO ROLE <SERVICE_OWNER_ROLE>;

GRANT READ ON SECRET <APP_DB>.<APP_SCHEMA>.STTM_BUILDER_OAUTH_SESSION_KEYS
  TO ROLE <SERVICE_OWNER_ROLE>;

GRANT READ ON SECRET <APP_DB>.<APP_SCHEMA>.STTM_BUILDER_OAUTH_CLIENT_CREDENTIALS TO ROLE <SERVICE_OWNER_ROLE>;
GRANT READ ON SECRET <APP_DB>.<APP_SCHEMA>.STTM_BUILDER_OAUTH_SESSION_KEYS TO ROLE <SERVICE_OWNER_ROLE>;
