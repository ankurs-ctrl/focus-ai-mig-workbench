## 1. Legacy relation-alias resolution gap

**Symptom:** mappings saved before aliases became identity-derived (e.g.
`servicing_advisors_15`, a positional index suffix) silently failed to
resolve after reload — the compiler and the frontend both regenerate aliases
from relation *identity* now (`servicing_advisors_kc2ar1`, a 6-char hash), so
the old qualifier matches nothing.

**Fix:** resolve the old qualifier structurally (by table name / stem) against
the current relation graph, and only accept the match if it's unambiguous.

### `services/sttm-builder/app/core/mapping_sql.py`

```python
@staticmethod
def _alias_stem(token: str) -> str:
    """Strip a generated alias suffix, leaving the durable name stem.

    Relation aliases have always had the shape ``<stem>_<suffix>``. The
    suffix used to be the array index (``servicing_advisors_15``) and is now
    an identity digest (``servicing_advisors_kc2ar1``). Both schemes share
    the stem, so the stem is the only stable key for matching a reference
    stored under one scheme against an alias generated by the other.

    Stripping is deliberately lossy: a table genuinely ending in a short
    word (``CONTACT_UDF_FIELDS``) also matches the suffix shape. Callers
    must therefore try the unstripped token first and treat an ambiguous
    stem match as unresolved.
    """
    text = str(token or "").strip().lower()
    stem, separator, suffix = text.rpartition("_")
    if separator and stem and re.fullmatch(r"\d+|[0-9a-z]{6}", suffix):
        return stem
    return text

@staticmethod
def _resolve_legacy_alias_relation(
    qualifier: str,
    nodes: dict[str, RelationNode],
    alias_to_id: dict[str, str],
) -> str | None:
    """Resolve a reference stored under a superseded alias scheme.

    Mappings saved before aliases became identity-derived persisted the
    positional form. Those stored references cannot be rewritten in place
    without a migration, so resolve them structurally here and fail closed
    whenever the match is not unique.
    """
    candidates: dict[str, set[str]] = {}

    def register(key: str | None, relation_id: str) -> None:
        token = str(key or "").strip().lower()
        if token:
            candidates.setdefault(token, set()).add(relation_id)

    for relation_id, node in nodes.items():
        alias = None
        for candidate_alias, candidate_id in alias_to_id.items():
            if candidate_id == relation_id:
                alias = candidate_alias
                break
        register(alias, relation_id)
        register(MappingSqlService._alias_stem(alias or ""), relation_id)
        if node.table and node.table.table:
            register(node.table.table, relation_id)
        if node.physical_view_name:
            register(node.physical_view_name.rsplit(".", 1)[-1], relation_id)
        register(relation_id.rsplit(".", 1)[-1], relation_id)

    # Exact match before stem match: an unstripped token is unambiguous
    # evidence, while a stem can collide with a legitimately suffixed name.
    for key in (qualifier, MappingSqlService._alias_stem(qualifier)):
        matches = candidates.get(str(key or "").strip().lower(), set())
        if len(matches) == 1:
            return next(iter(matches))
    return None
```

And into **expression** normalization (dependencies weren't the only place a
stale alias could appear — the mapping `expression` string references
qualifiers too):

```python
@staticmethod
def _normalize_legacy_expression_aliases(
    expression: str,
    nodes: dict[str, RelationNode],
    alias_to_id: dict[str, str],
    aliases: dict[str, str],
) -> str:
    """Rewrite superseded alias qualifiers onto their current aliases.

    Namespace normalization only recognises current aliases, relation IDs
    and physical names. A mapping saved under the positional alias scheme
    carries none of those, so its qualifiers survive normalization and would
    be rejected by alias validation. Resolve each unknown qualifier
    structurally and rewrite it; unresolvable qualifiers are left untouched
    so validation still reports them.
    """
    scrubbed = re.sub(r"'(?:''|[^'])*'", "''", expression)
    known = set(alias_to_id)
    known.update(node.relation_id.upper() for node in nodes.values())
    normalized = expression
    rewritten: set[str] = set()
    for qualifier, _column in re.findall(
        r"(?<![$\w])([A-Za-z_][A-Za-z0-9_$]*)\s*\.\s*([A-Za-z_][A-Za-z0-9_$]*)",
        scrubbed,
    ):
        if qualifier.upper() in known or qualifier in rewritten:
            continue
        relation_id = MappingSqlService._resolve_legacy_alias_relation(
            qualifier, nodes, alias_to_id
        )
        if relation_id is None:
            continue
        rewritten.add(qualifier)
        normalized = re.sub(
            rf"(?<![A-Za-z0-9_$]){re.escape(qualifier)}\.",
            f"{aliases[relation_id]}.",
            normalized,
        )
    return normalized
```

### `frontend/src/features/sttm/mapping/relation-alias.ts` (new exports)

```ts
/**
 * Map superseded aliases onto the aliases the generator produces today.
 *
 * Mappings saved while aliases were positional persisted qualifiers such as
 * `servicing_advisors_15`. Physical-table aliases are regenerated from identity
 * on every load, so those stored qualifiers now resolve to nothing. Derived
 * sources restore their persisted alias verbatim and therefore never drift.
 *
 * The persisted relation graph is the authoritative record of which alias a
 * stored reference was written against, so pair each persisted node alias with
 * the alias its identity yields now. Entries are only emitted where the two
 * actually differ.
 */
export function buildLegacyAliasRewrites(
  nodes: Array<{ alias?: unknown; identity?: unknown; displaySeed?: unknown }>,
): Map<string, string> {
  const rewrites = new Map<string, string>();
  const ambiguous = new Set<string>();
  for (const node of nodes) {
    const storedAlias = String(node.alias ?? "").trim();
    const identity = String(node.identity ?? "").trim();
    if (!storedAlias || !identity) continue;
    const displaySeed = String(node.displaySeed ?? "").trim() || null;
    const currentAlias = relationAliasFor(identity, displaySeed);
    if (currentAlias === storedAlias) continue;
    const key = storedAlias.toUpperCase();
    const existing = rewrites.get(key);
    if (existing && existing !== currentAlias) {
      // One stored alias cannot stand for two relations. Guessing would
      // silently rewrite a reference onto the wrong table.
      ambiguous.add(key);
      continue;
    }
    rewrites.set(key, currentAlias);
  }
  for (const key of ambiguous) rewrites.delete(key);
  return rewrites;
}

/**
 * Rewrite `<legacyAlias>.COLUMN` qualifiers onto current aliases.
 *
 * Only a single-token qualifier is rewritten. A dotted reference is already a
 * fully qualified name, which the compiler resolves directly, so touching it
 * would risk corrupting a valid reference.
 */
export function rewriteLegacyAliasReference(
  value: string,
  rewrites: Map<string, string>,
): string {
  if (!rewrites.size) return value;
  const text = String(value ?? "");
  if (!text.trim()) return text;
  return text.replace(
    /(?<![A-Za-z0-9_$.])([A-Za-z_][A-Za-z0-9_$]*)(?=\s*\.\s*[A-Za-z_])/g,
    (match, qualifier: string) => rewrites.get(qualifier.toUpperCase()) ?? match,
  );
}
```

## 2. Alias-vocabulary unification (derived sources) + malformed preview SQL

**Symptom:** three different systems disagreed on a relation's alias —
`tableAlias()` (bare lowercase table name), `buildRelationshipQuery`'s
unaliased FQNs, and the hash aliases used in stored dependencies. This
produced structurally invalid preview SQL (e.g. a `WITH` CTE body glued
*after* a `SELECT * FROM ...`, because `normalizeBaseQuerySql` only ever
stripped a leading `SELECT *` and returned everything else — including any
leading `WITH` block — as if it were a bare `FROM` clause).

**Fix, part A — one alias vocabulary everywhere:** every relation (physical
or derived) is now aliased via `relationAliasFor()`, and every builder that
declares a `FROM`/`JOIN` now emits `AS <alias>` explicitly.

**Fix, part B — a paren/quote/comment-aware SQL splitter**, replacing the old
regex-only `normalizeBaseQuerySql`:

```ts
/**
 * Index of a top-level keyword, or -1.
 *
 * Quotes, comments and parenthesised subqueries are skipped, because a CTE body
 * contains its own SELECT/FROM (and may contain a nested WITH). A regex cannot
 * make that distinction, and treating a nested keyword as top level is what
 * produced structurally invalid preview SQL.
 */
function findTopLevelKeyword(sql: string, keyword: string, startIndex = 0): number {
  const upper = sql.toUpperCase();
  const target = keyword.toUpperCase();
  let depth = 0;
  let index = startIndex;
  while (index < sql.length) {
    const character = sql[index];
    if (character === "'" || character === '"') {
      const quote = character;
      index += 1;
      while (index < sql.length) {
        if (sql[index] === quote) {
          if (sql[index + 1] === quote) { index += 2; continue; }
          index += 1;
          break;
        }
        index += 1;
      }
      continue;
    }
    if (character === '-' && sql[index + 1] === '-') {
      const newline = sql.indexOf('\n', index);
      index = newline === -1 ? sql.length : newline + 1;
      continue;
    }
    if (character === '/' && sql[index + 1] === '*') {
      const end = sql.indexOf('*/', index + 2);
      index = end === -1 ? sql.length : end + 2;
      continue;
    }
    if (character === '(') { depth += 1; index += 1; continue; }
    if (character === ')') { depth -= 1; index += 1; continue; }
    if (depth === 0 && upper.startsWith(target, index)) {
      const before = index === 0 ? '' : sql[index - 1];
      const after = sql[index + target.length] ?? '';
      if (!/[A-Za-z0-9_$]/.test(before) && !/[A-Za-z0-9_$]/.test(after)) {
        return index;
      }
    }
    index += 1;
  }
  return -1;
}

/**
 * Split a source query into its leading CTE block and its FROM tail.
 *
 * Callers rebuild a query by substituting their own projection, so they need the
 * two parts separately: a `WITH` block must stay in front of the new SELECT,
 * while the FROM tail follows it. The previous implementation only stripped a
 * leading `SELECT *` and therefore returned an entire `WITH ... SELECT * FROM
 * ...` query as if it were a FROM clause, which callers then spliced in after
 * their projection.
 */
function splitBaseQuerySql(baseSql: string): { withBlock: string; fromTail: string } {
  const trimmed = String(baseSql ?? '').trim().replace(/;+\s*$/, '');
  if (!trimmed) return { withBlock: '', fromTail: '' };

  let withBlock = '';
  let remainder = trimmed;
  if (/^WITH\b/i.test(trimmed)) {
    const selectIndex = findTopLevelKeyword(trimmed, 'SELECT');
    if (selectIndex > 0) {
      withBlock = trimmed.slice(0, selectIndex).trimEnd();
      remainder = trimmed.slice(selectIndex);
    }
  }

  // A bare `FROM ...` tail is already in the shape callers need.
  if (!/^SELECT\b/i.test(remainder)) {
    return { withBlock, fromTail: remainder };
  }

  const fromIndex = findTopLevelKeyword(remainder, 'FROM');
  return { withBlock, fromTail: fromIndex === -1 ? '' : remainder.slice(fromIndex) };
}
```

`buildSourceQueryPreviewSql` / `buildMappingSelectSql` now emit
`withBlock` (if any) → `SELECT <projection>` → `fromTail`, instead of
`SELECT ... <mangled fromClause>`:


## 3. nginx body-size limit + CORS-losing exception path

**Symptom:** SQL upload / large workspace-snapshot posts intermittently failed
with a bare "Network Error" (no CORS headers on the response at all, so the
browser couldn't even report a status code).

**Root cause A — nginx default body cap.** `client_max_body_size` was never
set anywhere in the repo, so nginx defaulted to 1 MB. Uploads accept SQL files
up to 5 MB, and the snapshot posted afterward re-serializes that SQL (plus
every CTE body), multiplying the payload 2-3x.

**Root cause B — `ClientDisconnect` escaping past CORS.** When a client
disconnects mid-body-read, `GuardrailsMiddleware` let `ClientDisconnect`
propagate up through `ServerErrorMiddleware`, which sits **outside**
`CORSMiddleware` — so its response carries no `Access-Control-Allow-Origin`
header and the browser reports an opaque network failure instead of the real
status.

### `nginx/nginx.conf`

```nginx
http {
  # ...
  sendfile on;
  keepalive_timeout 65;

  # Uploads accept SQL files up to 5 MB, and the workspace snapshot posted
  # afterwards re-serialises that SQL (plus every CTE body), so the request body
  # runs several times the source file. nginx defaults to 1 MB and rejects the
  # rest with a 413 that the browser surfaces as an opaque network failure.
  client_max_body_size 64m;
  client_body_buffer_size 256k;

  gzip on;
  # ...
}
```

## 4. SQL-upload preview key mismatch

**Symptom:** the "Review parsed SQL" panel showed an em dash for every table
reference instead of a name.

**Root cause:** the backend emits `raw_identifier` for an unresolved/ambiguous
reference (no `resolved_fqn`); the panel read `item.raw_reference`, a key the
payload never contains.

### `frontend/src/features/sttm/mapping/sql-bundle-review-panel.tsx`

```ts
/**
 * Display name for a parsed table reference.
 *
 * An unresolved or ambiguous reference has no `resolved_fqn`, so fall back to the
 * identifier as written in the uploaded SQL. The resolver emits that under
 * `raw_identifier`; reading only a `raw_reference` key that the payload never
 * contained rendered every such row as an em dash.
 */
function referenceLabel(item: Record<string, unknown>): string {
  return label(item.resolved_fqn ?? item.raw_identifier ?? item.raw_reference);
}

/** Name the competing matches that make a reference ambiguous. */
function candidateHint(item: Record<string, unknown>): string {
  if (String(item.resolution_status ?? "") !== "ambiguous") return "";
  const candidates = Array.isArray(item.candidate_fqns)
    ? item.candidate_fqns.map((value) => String(value)).filter(Boolean)
    : [];
  if (!candidates.length) return "";
  return ` · matches ${candidates.join(", ")}`;
}
```

## 5. Disconnected-relations error message (STTM 2901 diagnosis)

**Symptom:** compiling STTM 2901 rejected with "Required relations are
disconnected from the driving relation" and only opaque physical view names
(e.g. `DSV_FIR_DERIVED_...`), which looked like a bug but is **correct
behavior** — several selected derived sources genuinely have no join edge
back to the driving `CONTACTS` relation.

**Fix (not a bypass — a better error):** name the alias alongside the
physical label, and for a derived source whose disconnection is caused by its
own unjoined base table(s), name those tables too.

### `services/sttm-builder/app/core/mapping_sql.py`

```python
if disconnected:
    # A derived source is reported by its physical view name, which is
    # opaque (DSV_FIR_DERIVED_...). Name the alias as well, and say when a
    # derived source is unreachable only because its own base table is
    # unjoined - that is the edge the author actually has to add.
    joined_relations = connected_ids
    disconnected_labels = []
    for relation_id in sorted(disconnected):
        node = nodes[relation_id]
        label = (
            node.table.qualified_name
            if node.table is not None
            else node.physical_view_name or node.relation_id
        )
        alias = aliases.get(relation_id)
        if alias and alias not in label:
            label = f"{label} ({alias})"
        unjoined_bases = [
            (
                nodes[base_id].table.qualified_name
                if nodes[base_id].table is not None
                else base_id
            )
            for base_id in node.base_relation_ids
            if base_id in nodes
            and base_id not in joined_relations
            and base_id != relation_id
        ]
        if unjoined_bases:
            label = f"{label} - built on unjoined " + ", ".join(sorted(unjoined_bases))
        disconnected_labels.append(label)
    raise ValueError(
        "Required relations are disconnected from the driving relation: "
        + "; ".join(disconnected_labels)
        + ". Add validated relation edges from the driving graph to these relations, "
        "or remap the affected targets to outputs already produced by a connected derived source."
    )
```

(Only the label formatting and the join separator — `"; "` instead of the
old `", "` — changed; the rejection logic itself is unchanged and correct.)

## 7. Cortex Analyst 502 — quoted logical column names (392700)

**Symptom:** unrelated to the plan above — surfaced mid-session as "Cortex
Analyst request failed" in the UI, no further detail.

**Root cause:** Cortex Analyst requires a semantic-model logical `name` to be
a bare identifier (letters/digits/underscore/`$`, starting with a letter or
underscore). `expr` may be quoted to preserve casing. The YAML composer wrote
the *quoted* identifier into `name` for mixed-case physical columns (e.g.
`name: '"BillingCity"'`), which fails the **entire** semantic view with
Snowflake error 392700 — one bad column takes down the whole model.
Confirmed present in 31 of 231 `ACTIVE` rows in
`SCH_STTM_MCP.SEM_NATIVE_VIEWS.CA_YAML_MODEL`.

### `services/sttm-builder/app/core/semantic_context.py`

```python
def _strip_identifier_quotes(value: str) -> str:
    text = str(value).strip()
    if len(text) >= 2 and text.startswith('"') and text.endswith('"'):
        return text[1:-1].replace('""', '"')
    return text


def _normalize_logical_names(node: Any) -> None:
    """Force every logical name in a composed model to a bare identifier.

    Cortex Analyst requires a logical name to start with a letter or underscore
    and contain only letters, digits, underscores or dollar signs. Registry
    fragments for mixed-case physical columns carry the *quoted* identifier as the
    logical name (``name: '"BillingCity"'``), and a single such column fails the
    whole model with error 392700 - which surfaces only as a generic Analyst
    request failure.

    ``expr`` is deliberately left alone: it is the physical reference and needs
    its quoting to preserve casing. Stripping the quotes from ``name`` also makes
    these models self-consistent, because reference lists such as
    ``primary_key.columns`` already use the bare name.
    """
    if isinstance(node, dict):
        for key, value in node.items():
            if key == "name" and isinstance(value, str):
                node[key] = _strip_identifier_quotes(value)
            else:
                _normalize_logical_names(value)
    elif isinstance(node, list):
        for item in node:
            _normalize_logical_names(item)
```

Called immediately before `yaml.safe_dump` in `_compose_authoritative_yaml`:

```python
_normalize_logical_names(payload)
rendered = yaml.safe_dump(payload, sort_keys=False, allow_unicode=False)
```

**⚠️ Important — this fix does NOT retroactively repair existing bad rows.**
`_compose_authoritative_yaml` only runs when a semantic view is
promoted/rebuilt; it is not re-run on every Analyst request. Verified in this
environment **after** deploying 2.9.1:

```sql
SELECT COUNT(*) FROM FFP_HDP_DLAB_DB_DEV.SCH_STTM_MCP.SEM_NATIVE_VIEWS
WHERE STATUS = 'ACTIVE' AND CA_YAML_MODEL ILIKE '%name: ''"%';
-- still returns 31
```

**Action needed in every environment (including this one):** the 31
already-broken views will keep failing Analyst requests until each is
re-promoted through the normal build flow, or until someone runs a one-time
text patch against the stored `CA_YAML_MODEL` column. This was **not** done
yet in this environment — flag it as an open follow-up, not a completed fix,
when syncing.

```python
def _stream_error_envelope(
    req, title: str, status: int, code: str, detail: str | None = None,
) -> dict[str, Any]:
    # Repeating the title as the detail discarded the only description of what
    # actually went wrong, leaving the cause visible solely in service logs.

    }
```

## 10. Data-only fix — NOT a code change, check if needed in the target env

Separately from the plan above, this session also found and backfilled
historical rows in `TBL_STTM` where `LAST_MODIFIED_BY` / `ACTOR_USER_ID` /
`STTM_METADATA:actor_user_id` stored a numeric user ID instead of the
resolved display name (visible in the Mappings dashboard as a number instead
of a username). The **code path that writes these fields was already
correct** — every bad row predates this environment's most recent deploy,
with zero counterexamples created after. If the target environment has the
same symptom, run the equivalent read-only check first:

```sql
SELECT COUNT(*) FROM <DB>.SCH_STTM_METADATA.TBL_STTM
WHERE REGEXP_LIKE(LAST_MODIFIED_BY, '^[0-9]+$')
   OR REGEXP_LIKE(ACTOR_USER_ID, '^[0-9]+$')
   OR REGEXP_LIKE(STTM_METADATA:actor_user_id::string, '^[0-9]+$');
```

If it returns > 0, the backfill run in this environment was:

```sql
UPDATE <DB>.SCH_STTM_METADATA.TBL_STTM s
SET LAST_MODIFIED_BY = u.DISPLAY_NAME
FROM <DB>.SCH_STTM_METADATA.TBL_USERS u
WHERE s.LAST_MODIFIED_BY IS NOT NULL
  AND REGEXP_LIKE(s.LAST_MODIFIED_BY, '^[0-9]+$')
  AND TO_VARCHAR(u.USER_ID) = s.LAST_MODIFIED_BY;

UPDATE <DB>.SCH_STTM_METADATA.TBL_STTM s
SET ACTOR_USER_ID = u.DISPLAY_NAME
FROM <DB>.SCH_STTM_METADATA.TBL_USERS u
WHERE s.ACTOR_USER_ID IS NOT NULL
  AND REGEXP_LIKE(s.ACTOR_USER_ID, '^[0-9]+$')
  AND TO_VARCHAR(u.USER_ID) = s.ACTOR_USER_ID;

UPDATE <DB>.SCH_STTM_METADATA.TBL_STTM s
SET STTM_METADATA = OBJECT_INSERT(s.STTM_METADATA, 'actor_user_id', u.DISPLAY_NAME::VARIANT, TRUE)
FROM <DB>.SCH_STTM_METADATA.TBL_USERS u
WHERE s.STTM_METADATA:actor_user_id IS NOT NULL
  AND REGEXP_LIKE(s.STTM_METADATA:actor_user_id::string, '^[0-9]+$')
  AND TO_VARCHAR(u.USER_ID) = s.STTM_METADATA:actor_user_id::string;
```

Use three independent joins (one per field), not one shared `COALESCE` join —
a shared join silently skips rows where one field is already a name but a
different field on the same row is still numeric.
